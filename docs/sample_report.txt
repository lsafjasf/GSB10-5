-- 值等价、写法不同（不作为差异） --
  - ALLOWED_IPS  [列表顺序/分隔符]
      prod = 10.0.0.1, 10.0.0.2,10.0.0.3
      staging = 10.0.0.3;10.0.0.1;10.0.0.2
  - FEATURE_FAST_PATH  [布尔写法]
      prod = on
      staging = true
  - HEAP_SIZE  [单位写法]
      prod = 512MB
      staging = 0.5GB
  - REGION  [空白字符、大小写]
      prod = cn-north
      staging = CN-NORTH  
  - STARTUP_TIMEOUT  [单位写法]
      prod = 30s
      staging = 30000ms
  - THREADS  [数值写法]
      prod = 8
      staging = 8.0

-- 有意差异（intentional）--
  - CACHE_TTL
      prod = 300
      staging = 600
      归因：whitelist | 依据：白名单允许：预发环境压测专用缓存策略
  - DB_HOST
      prod = db-prod.internal
      staging = db-staging.internal
      归因：expected | 依据：符合声明的期望值（DB_HOST 按环境应有不同）
  - EXPERIMENTAL_FLAG
      prod = off
      staging = <缺失>
      归因：whitelist | 依据：staging 允许不配置：白名单允许
  - LOG_LEVEL
      prod = INFO
      staging = DEBUG
      归因：expected | 依据：符合声明的期望值（LOG_LEVEL 按环境应有不同）

-- 待确认（to_confirm）--
  - EXTRA_DEBUG
      prod = <缺失>
      staging = 1
      归因：none | 依据：仅单环境定义，prod 缺失，无任何归因依据
  - TLS_CERT
      prod = ****** (sha256:9360c0c92c6f)
      staging = ****** (sha256:90321d367bb3)
      归因：none | 依据：两边值不同，白名单、声明期望值、历史基线均未覆盖

-- 疑似遗漏（suspected_omission）--
  - DB_MAX_CONN
      prod = 200
      staging = 100

  - NEW_RELIC_ENABLED
      prod = <缺失>
      staging = <缺失>
      归因：expected | 依据：声明应存在，但两个环境均未配置
